cmake_minimum_required(VERSION 3.24)
project(ShinyCore VERSION 0.2.0 LANGUAGES C CXX)
# Upstream libraries keep their C ABI; all ShinyCore code is modern C++23.
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>\n#include <memory>\nint main() { std::expected<std::unique_ptr<int>, int> v = std::make_unique<int>(23); return **v != 23; }" SHINY_HAS_CXX23_EXPECTED)
if(NOT SHINY_HAS_CXX23_EXPECTED)
  message(FATAL_ERROR "ShinyCore requires a C++23 compiler and standard library with std::expected. Use a current Clang/libc++, GCC/libstdc++, or MSVC toolchain.")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SHINY_GRAPHICS "Build the native window/audio backend" ON)
option(SHINY_NETWORK "Build the native ENet multiplayer module" OFF)
option(SHINY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(FetchContent)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()

# Pinned release archives with integrity checks; no system Lua ABI dependency.
FetchContent_Declare(lua
  URL https://www.lua.org/ftp/lua-5.4.9.tar.gz
  URL_HASH SHA256=2335b6c582a52654f94612bf10d2f4672805d05329aa6568b1d8cd9e5c6fb8e6)
FetchContent_MakeAvailable(lua)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${lua_SOURCE_DIR}/src/*.c")
list(REMOVE_ITEM LUA_SOURCES "${lua_SOURCE_DIR}/src/lua.c" "${lua_SOURCE_DIR}/src/luac.c")
add_library(shiny_lua STATIC ${LUA_SOURCES})
target_include_directories(shiny_lua PUBLIC "${lua_SOURCE_DIR}/src")
if(UNIX)
  target_link_libraries(shiny_lua PUBLIC m)
endif()

set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(box2d
  URL https://github.com/erincatto/box2d/archive/refs/tags/v3.1.1.tar.gz
  URL_HASH SHA256=fb6ef914b50f4312d7d921a600eabc12318bb3c55a0b8c0b90608fa4488ef2e4)
FetchContent_MakeAvailable(box2d)
add_library(shiny_core STATIC src/core.cpp src/physics.cpp src/input.cpp)
target_link_libraries(shiny_core PRIVATE box2d)
target_include_directories(shiny_core PUBLIC include)
if(UNIX)
  target_link_libraries(shiny_core PUBLIC m)
endif()
FetchContent_Declare(yyjson
  URL https://github.com/ibireme/yyjson/archive/refs/tags/0.12.0.tar.gz
  URL_HASH SHA256=b16246f617b2a136c78d73e5e2647c6f1de1313e46678062985bdcf1f40bb75d)
FetchContent_MakeAvailable(yyjson)
FetchContent_Declare(stb_truetype
  URL https://raw.githubusercontent.com/raysan5/raylib/5.5/src/external/stb_truetype.h
  URL_HASH SHA256=ecd30b05e0dd4fea3a13c26810dd9e1992dc379049482c393d5a19e6b5090aab
  DOWNLOAD_NO_EXTRACT TRUE)
FetchContent_MakeAvailable(stb_truetype)
add_library(shiny_text STATIC src/text.cpp)
target_include_directories(shiny_text PUBLIC include)
target_include_directories(shiny_text SYSTEM PRIVATE "${stb_truetype_SOURCE_DIR}")
add_library(shiny_script STATIC src/script.cpp src/input_replay.cpp src/script_data.cpp src/script_physics.cpp src/script_audio.cpp src/state.cpp src/project.cpp)
target_link_libraries(shiny_script PUBLIC shiny_text)
target_link_libraries(shiny_script PUBLIC shiny_core shiny_lua PRIVATE yyjson)
add_executable(shiny src/main.cpp)
target_link_libraries(shiny PRIVATE shiny_script)
if(MINGW)
  target_link_options(shiny PRIVATE -static "$<$<CONFIG:Release>:-s>")
endif()

if(SHINY_GRAPHICS)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
  # raylib's implicit F12 / Ctrl-F12 handlers must not consume game keys.
  # The host's --capture uses ExportImage directly and remains available.
  set(SUPPORT_SCREEN_CAPTURE OFF CACHE BOOL "" FORCE)
  set(SUPPORT_GIF_RECORDING OFF CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RMODELS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib
    URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz
    URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
  FetchContent_MakeAvailable(raylib)
  target_sources(shiny PRIVATE src/render.cpp)
  target_link_libraries(shiny PRIVATE raylib)
  target_compile_definitions(shiny PRIVATE SC_HAS_GRAPHICS=1)
endif()

set(SHINY_TARGETS shiny_core shiny_script shiny_text shiny)
if(SHINY_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "SHINY_SANITIZERS requires Clang/GCC with both ASan and UBSan runtimes; use the Linux sanitizer CI configuration.")
  endif()
  set(CMAKE_REQUIRED_FLAGS "-fsanitize=address,undefined")
  check_cxx_source_compiles("int main() { return 0; }" SHINY_HAS_SANITIZERS)
  unset(CMAKE_REQUIRED_FLAGS)
  if(NOT SHINY_HAS_SANITIZERS)
    message(FATAL_ERROR "ASan/UBSan runtimes are missing from this toolchain. Disable SHINY_SANITIZERS or use the Linux sanitizer CI configuration.")
  endif()
  foreach(dependency shiny_lua box2d yyjson)
    target_compile_options(${dependency} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
endif()
if(SHINY_NETWORK)
  FetchContent_Declare(enet
    URL https://github.com/lsalzman/enet/archive/refs/tags/v1.3.18.tar.gz
    URL_HASH SHA256=28603c895f9ed24a846478180ee72c7376b39b4bb1287b73877e5eae7d96b0dd)
  FetchContent_MakeAvailable(enet)
  if(WIN32)
    target_link_libraries(enet winmm ws2_32)
  endif()
  add_library(shiny_net STATIC src/net.cpp)
  target_include_directories(shiny_net PUBLIC include PRIVATE "${enet_SOURCE_DIR}/include")
  target_link_libraries(shiny_net PRIVATE enet)
  target_sources(shiny_script PRIVATE src/net_lua.cpp)
  target_link_libraries(shiny_script PUBLIC shiny_net)
  target_compile_definitions(shiny_script PRIVATE SC_HAS_NETWORK=1)
  list(APPEND SHINY_TARGETS shiny_net)
  if(SHINY_SANITIZERS AND NOT MSVC)
    target_compile_options(enet PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_input tests/test_input.cpp)
  target_link_libraries(test_input PRIVATE shiny_core)
  add_test(NAME input COMMAND test_input)
  list(APPEND SHINY_TARGETS test_input)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE shiny_core)
  add_test(NAME core COMMAND test_core)
  add_executable(test_script tests/test_script.cpp)
  target_link_libraries(test_script PRIVATE shiny_script)
  add_test(NAME script COMMAND test_script)
  set_tests_properties(script PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  list(APPEND SHINY_TARGETS test_core test_script)
  if(SHINY_NETWORK)
    add_executable(test_net tests/test_net.cpp)
    target_link_libraries(test_net PRIVATE shiny_net)
    add_test(NAME net COMMAND test_net)
    add_executable(test_net_lua tests/test_net_lua.cpp)
    target_link_libraries(test_net_lua PRIVATE shiny_script)
    add_test(NAME net_lua COMMAND test_net_lua)
    list(APPEND SHINY_TARGETS test_net test_net_lua)
    set_tests_properties(net net_lua PROPERTIES TIMEOUT 30)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME input_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/input_integration.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME features COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/features.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME realtime COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/realtime.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
    if(SHINY_NETWORK)
      add_test(NAME network_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/network_integration.py" "$<TARGET_FILE:shiny>" "${CMAKE_CURRENT_SOURCE_DIR}")
      set_tests_properties(network_integration PROPERTIES TIMEOUT 30)
    endif()
  endif()
endif()
foreach(target IN LISTS SHINY_TARGETS)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(SHINY_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()

install(FILES docs/input.md docs/api.lua docs/llm-guide.md DESTINATION share/shiny/docs)
install(DIRECTORY examples/workshop DESTINATION share/shiny/examples)
if(SHINY_NETWORK)
  install(DIRECTORY examples/duet DESTINATION share/shiny/examples)
  install(FILES docs/networking.md DESTINATION share/shiny/docs)
endif()
install(FILES LICENSE THIRD_PARTY.md DESTINATION share/shiny)
install(DIRECTORY licenses DESTINATION share/shiny)
